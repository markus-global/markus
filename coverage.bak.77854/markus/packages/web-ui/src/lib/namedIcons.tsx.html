/**
 * Best-effort icon name for a built-in skill that carries no icon of its own.
 * Matches by known slug first, then category, else a neutral default.
 */
export function iconForSkill(nameOrSlug?: string, category?: string): string {
  const key = (nameOrSlug || '').toLowerCase();
  if (SKILL_ICON_BY_NAME[key]) return SKILL_ICON_BY_NAME[key];
  if (category && SKILL_ICON_BY_CATEGORY[category]) return SKILL_ICON_BY_CATEGORY[category];
  return 'layers';
}
