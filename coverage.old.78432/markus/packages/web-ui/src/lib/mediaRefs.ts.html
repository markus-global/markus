/**

export function toAbsoluteHubUrl(url: string, hubUrl: string): string {
  if (/^https?:\/\//i.test(url)) return url;
  const origin = hubUrl.replace(/\/+$/, '');
  return `${origin}${url.startsWith('/') ? '' : '/'}${url}`;
